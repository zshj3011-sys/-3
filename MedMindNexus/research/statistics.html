# Step 1: 加载数据与包 library(survival); library(survminer); library(MatchIt) df <- read.csv("cad_t2dm_cohort.csv") # Step 2: 基线特征比较 library(tableone) vars <- c("age", "sex", "bmi", "ldl", "hba1c", "smoking") tab1 <- CreateTableOne(vars=vars, strata="group", data=df, test=TRUE) print(tab1, smd=TRUE) # Step 3: Kaplan-Meier 生存曲线 km <- survfit(Surv(time, mace) ~ group, data=df) ggsurvplot(km, data=df, pval=TRUE, conf.int=TRUE, risk.table=TRUE, xlab="Months", ylab="MACE-free survival") # Step 4: Cox 比例风险回归（多变量调整） cox <- coxph(Surv(time, mace) ~ group + age + sex + ldl + hba1c + smoking, data=df) summary(cox) # Step 5: 检验 PH 假设 cox.zph(cox) # Step 6: 倾向评分匹配（敏感性分析） m <- matchit(group ~ age + sex + bmi + ldl + hba1c, data=df, method="nearest", ratio=1) matched_df <- match.data(m) cox_psm <- coxph(Surv(time, mace) ~ group, data=matched_df) summary(cox_psm)

Statistical Analysis.
Continuous variables were expressed as mean ± standard deviation or median (interquartile range) and compared using Student's t-test or the Mann-Whitney U test as appropriate. Categorical variables were expressed as numbers (percentages) and compared using the chi-square test or Fisher's exact test.

The cumulative incidence of MACE was estimated using the Kaplan-Meier method, and between-group differences were assessed by the log-rank test. Multivariable Cox proportional hazards regression was used to estimate hazard ratios (HRs) and 95% confidence intervals (CIs) for MACE, adjusting for age, sex, baseline LDL-C, HbA1c, and smoking status. The proportional hazards assumption was verified using scaled Schoenfeld residuals.

To minimize confounding by indication, propensity score matching (1:1 nearest neighbor without replacement) was performed as a sensitivity analysis. Two-sided P values < 0.05 were considered statistically significant. All analyses were performed using R version 4.4.0.
